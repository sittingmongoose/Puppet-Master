```bash
python3 -B materialize.py \
  --destination /absolute/caller/NEW-dedicated-copy \
  --authorized-provider-instance ACTUAL_AUTHORIZED_PROVIDER_ID \
  --brief /absolute/caller/NEW-brief.md --plan /absolute/caller/NEW-plan.md \
  --run-id NEW-unique-id --method 'A1+A8' \
  --activation-capture-version exact-capture-v21 \

```
